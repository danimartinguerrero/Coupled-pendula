\begin{tikzpicture}[remember picture, overlay]
  \coordinate (O) at (current page.south west);

  % ----------------------------------------------------------------

  \node[anchor=north west, inner sep=0] at ($(O)+(2.2cm,27.9cm)$)

  \node[anchor=north east, inner sep=0, text=grispizarra, font=\scshape\small]

  % ----------------------------------------------------------------
  % Motivo gráfico: los dos péndulos acoplados, vistos de frente, en un
  % instante del modo pulsante (el izquierdo con poca amplitud, el derecho
  % con toda). Arcos punteados: trayectoria de las masas lenticulares.
  % Las cotas van paralelas a las varillas y por fuera de ellas, y los
  % rótulos de los ángulos, al otro lado de la vertical, para no tapar nada.
  % ----------------------------------------------------------------
  \begin{scope}[shift={($(O)+(10.5cm,23.75cm)$)}, x=1cm, y=1cm]
    \def\Lp{6.55}\def\lp{2.95}\def\dsep{5.6}\def\phiA{-6}\def\phiB{12}
    % barra de soporte, muy discreta
    \draw[grisclaro, line width=2.2pt, line cap=round] (-4.6,0.95) -- (4.6,0.95);
    % trayectorias de las masas y escala angular tenue
    \foreach \cx in {-2.8,2.8}{
      \draw[azulclaro, line width=0.5pt, densely dotted]
        ($(\cx,0)+(-104:\Lp)$) arc[start angle=-104, end angle=-76, radius=\Lp];
      \foreach \a in {-100,-95,...,-80}{
        \draw[azulclaro!80, line width=0.3pt]
          ($(\cx,0)+(\a:\Lp+0.55)$) -- ($(\cx,0)+(\a:\Lp+0.72)$);}}
    % el par de péndulos
    \parpendulos{\phiA}{\phiB}{\lp}{\Lp}{\dsep}{1.2}
    % ángulos (rótulo al otro lado de la vertical de equilibrio)
    \angulo{P1}{\phiA}{1.9}{$\varphi_1$}
    \angulo{P2}{\phiB}{1.9}{$\varphi_2$}
    % longitud de acoplamiento: paralela a la varilla derecha, por fuera
    \draw[grispizarra, line width=0.45pt, |<->|]
      ($(P2)+({0.62*cos(\phiB)},{0.62*sin(\phiB)})$) --
      ($(S2)+({0.62*cos(\phiB)},{0.62*sin(\phiB)})$)
      node[midway, right=2pt, text=azulprof] {$\ell$};
    % longitud del péndulo: paralela a la varilla izquierda, por fuera de la masa
    \draw[grispizarra!80, line width=0.4pt, |<->|]
      ($(P1)+({-0.95*cos(\phiA)},{-0.95*sin(\phiA)})$) --
      ($(B1)+({-0.95*cos(\phiA)},{-0.95*sin(\phiA)})$)
      node[pos=0.55, left=2pt, text=azulprof] {$L$};
    % rótulos discretos
    \draw[grispizarra!70, line width=0.35pt] ($(S1)!0.5!(S2)+(0,-0.14)$) -- ++(0.9,-1.25);
    \fill[azulacero] ($(S1)!0.5!(S2)+(0,-0.14)$) circle (1.3pt);
    \node[anchor=west, text=grispizarra, font=\itshape\small]
      at ($(S1)!0.5!(S2)+(0.95,-1.34)$) {muelle, $k$};
    \draw[grispizarra!70, line width=0.35pt] ($(B2)+(0.5,0.12)$) -- ++(0.6,0.5);
    \node[anchor=west, text=grispizarra, font=\itshape\small]
      at ($(B2)+(1.12,0.62)$) {masa lenticular, $m$};
    \draw[grispizarra!70, line width=0.35pt] ($(P1)+(-0.3,0.5)$) -- ($(P1)+(-1.55,0.5)$);
    \node[anchor=east, text=grispizarra, font=\itshape\small]
      at ($(P1)+(-1.6,0.5)$) {eje de giro y sensor};
  \end{scope}

  % ----------------------------------------------------------------
  % Franja: la pulsación. Los dos péndulos se pasan la energía; cada
  % trazo es una portadora rápida dentro de una envolvente lenta, y las
  % envolventes están desfasadas un cuarto de ciclo.
  % ----------------------------------------------------------------
  \begin{scope}[shift={($(O)+(0,13.55cm)$)}, x=1cm, y=1cm]
    \def\xa{2.2}\def\xb{18.8}\def\wm{0.2838}\def\wp{6.6}
    \draw[grispizarra!35, line width=0.3pt] (\xa,0.62) -- (\xb,0.62);
    \draw[grispizarra!35, line width=0.3pt] (\xa,-0.62) -- (\xb,-0.62);
    % péndulo 2 (arriba): cos(w- t) cos(w+ t)
    \draw[azulacero!85, line width=0.5pt, densely dashed, domain=\xa:\xb, samples=120]
      plot (\x,{0.62+0.48*abs(cos(\wm*(\x-\xa) r))});
    \draw[azulprof, line width=0.5pt, domain=\xa:\xb, samples=1100]
      plot (\x,{0.62+0.48*cos(\wm*(\x-\xa) r)*cos(\wp*(\x-\xa) r)});
    % péndulo 1 (abajo): sin(w- t) sin(w+ t)
    \draw[turquesa!80!azulprof, line width=0.5pt, densely dashed, domain=\xa:\xb, samples=120]
      plot (\x,{-0.62+0.48*abs(sin(\wm*(\x-\xa) r))});
    \draw[turquesa!90!azulprof, line width=0.5pt, domain=\xa:\xb, samples=1100]
      plot (\x,{-0.62+0.48*sin(\wm*(\x-\xa) r)*sin(\wp*(\x-\xa) r)});
    \node[anchor=east, text=azulprof, font=\footnotesize] at (\xa-0.08,0.62) {$\varphi_2$};
    \node[anchor=east, text=turquesa!90!azulprof, font=\footnotesize] at (\xa-0.08,-0.62) {$\varphi_1$};
    % periodo de pulsación sobre la envolvente superior
    \draw[grispizarra, line width=0.4pt, |<->|]
      (\xa,1.32) -- ({\xa+pi/\wm},1.32)
      node[midway, above=0pt, text=azulprof, font=\footnotesize, fill=white, inner sep=1.2pt] {$T_{\mathrm{b}}=\pi/\omega_{-}$};
    \node[anchor=north, text=grispizarra, font=\scshape\scriptsize] at (10.5,-1.32)
      {\textls[160]{la energía pasa de un péndulo al otro}};
  \end{scope}
